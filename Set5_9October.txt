Q3 Use a parameterized cursor FOR LOOP with %ROWTYPE to filter students by class.
• DECLARE a cursor CURSOR c_class(p_class VARCHAR2) IS SELECT * FROM Student WHERE Class = p_class.
• Use a cursor FOR LOOP with a record variable declared as Student%ROWTYPE (or the loop's implicit record)
to iterate.
• Call it by passing a class value, e.g., '10B'

Code : 
    CREATE TABLE student(ID number, Name varchar2(20), marks number, class number);
    INSERT  INTO student VALUES(1,'a',50,10);
    INSERT  INTO student VALUES(2,'b',90,10);
    INSERT  INTO student VALUES(3,'c',80,12);

    SET SERVEROUTPUT ON;

    DECLARE
        record student%rowtype;
        CURSOR c_class(p_class varchar2) IS SELECT * FROM student WHERE class = p_class;
    BEGIN
        FOR record IN c_class(12) LOOP
            DBMS_OUTPUT.PUT_LINE('ID ' || record.ID || ' Name ' || record.Name || ' marks ' || record.marks || ' class ' || record.class);
            
            EXIT WHEN sql%NOTFOUND;
        END LOOP;
    END;
